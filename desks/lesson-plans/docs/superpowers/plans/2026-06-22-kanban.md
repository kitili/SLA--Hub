# Hackathon Kanban Board Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a projector-optimised drag-and-drop Kanban board at `/kanban` with state persisted in `kanban.json`.

**Architecture:** React page fetches cards from `GET /api/kanban` on mount; drag-and-drop via `@dnd-kit`; dropping a card calls `POST /api/kanban` which writes `kanban.json`; Vite HMR refreshes the projector when the file changes from direct edits.

**Tech Stack:** React 19, @dnd-kit/core + @dnd-kit/sortable + @dnd-kit/utilities, Express (existing), fs (Node built-in)

- [ ] **Step 2: Create `server/routes/kanban.js`**

```js
import express from 'express';
import path from 'path';
import fs from 'fs';
import { fileURLToPath } from 'url';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const KANBAN_PATH = path.resolve(__dirname, '../../kanban.json');
const router = express.Router();

router.get('/', (_req, res) => {
  try {
    res.json(JSON.parse(fs.readFileSync(KANBAN_PATH, 'utf8')));
  } catch {
    res.json({ cards: [] });
  }
});

router.post('/', (req, res) => {
  const { cards } = req.body;
  if (!Array.isArray(cards)) return res.status(400).json({ error: 'cards must be an array' });
  try {
    const data = JSON.parse(fs.readFileSync(KANBAN_PATH, 'utf8'));
    data.cards = cards;
    fs.writeFileSync(KANBAN_PATH, JSON.stringify(data, null, 2));
    res.json({ ok: true });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

export default router;
```

```jsx
import { useState, useEffect } from 'react';
import {
  DndContext,
  DragOverlay,
  closestCorners,
  PointerSensor,
  useSensor,
  useSensors,
} from '@dnd-kit/core';
import {
  SortableContext,
  useSortable,
  verticalListSortingStrategy,
} from '@dnd-kit/sortable';
import { useDroppable } from '@dnd-kit/core';
import { CSS } from '@dnd-kit/utilities';
import './Kanban.css';

const COLUMNS = [
  { id: 'ideas',       label: '💡 Ideas',       accent: '#a78bfa' },
  { id: 'in-progress', label: '🔨 In Progress', accent: '#fb923c' },
  { id: 'done',        label: '✅ Done',         accent: '#4ade80' },
];

const TAG_COLORS = {
  purple: { bg: '#7c3aed', text: '#e9d5ff' },
  cyan:   { bg: '#0e7490', text: '#a5f3fc' },
  green:  { bg: '#065f46', text: '#a7f3d0' },
  orange: { bg: '#9a3412', text: '#fed7aa' },
  red:    { bg: '#991b1b', text: '#fecaca' },
};

function KanbanCard({ card }) {
  const { attributes, listeners, setNodeRef, transform, transition, isDragging } =
    useSortable({ id: card.id });

  const style = {
    transform: CSS.Transform.toString(transform),
    transition,
    opacity: isDragging ? 0.3 : 1,
  };

  const tagColor = card.tag ? (TAG_COLORS[card.tag.color] ?? TAG_COLORS.purple) : null;

  return (
    <div ref={setNodeRef} style={style} className="kanban-card" {...attributes} {...listeners}>
      <span className="kanban-card-title">{card.title}</span>
      <div className="kanban-card-meta">
        {tagColor && (
          <span className="kanban-tag" style={{ background: tagColor.bg, color: tagColor.text }}>
            {card.tag.label}
          </span>
        )}
        {card.owner && <span className="kanban-owner">{card.owner}</span>}
      </div>
    </div>
  );
}

function KanbanColumn({ col, cards }) {
  const { setNodeRef, isOver } = useDroppable({ id: col.id });

  return (
    <div className="kanban-column">
      <div className="kanban-col-header" style={{ borderTopColor: col.accent }}>
        <span style={{ color: col.accent }}>{col.label}</span>
        <span className="kanban-col-count">{cards.length}</span>
      </div>
      <SortableContext items={cards.map((c) => c.id)} strategy={verticalListSortingStrategy}>
        <div ref={setNodeRef} className={`kanban-col-body${isOver ? ' kanban-col-over' : ''}`}>
          {cards.map((card) => (
            <KanbanCard key={card.id} card={card} />
          ))}
        </div>
      </SortableContext>
    </div>
  );
}

export default function Kanban() {
  const [cards, setCards] = useState([]);
  const [activeCard, setActiveCard] = useState(null);

  const sensors = useSensors(
    useSensor(PointerSensor, { activationConstraint: { distance: 8 } })
  );

  useEffect(() => {
    const load = () =>
      fetch('/api/kanban')
        .then((r) => r.json())
        .then((data) => setCards(data.cards ?? []));

    load();
    const interval = setInterval(load, 3000);
    return () => clearInterval(interval);
  }, []);

  function handleDragStart({ active }) {
    setActiveCard(cards.find((c) => c.id === active.id) ?? null);
  }

  function handleDragEnd({ active, over }) {
    setActiveCard(null);
    if (!over) return;

    const colIds = COLUMNS.map((c) => c.id);
    const targetColumn = colIds.includes(over.id)
      ? over.id
      : cards.find((c) => c.id === over.id)?.column;

    if (!targetColumn) return;

    const newCards = cards.map((c) =>
      c.id === active.id ? { ...c, column: targetColumn } : c
    );
    setCards(newCards);

    fetch('/api/kanban', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ cards: newCards }),
    });
  }

  return (
    <div className="kanban-root">
      <div className="kanban-header">
        <h1>Hackathon Board</h1>
      </div>
      <DndContext
        sensors={sensors}
        collisionDetection={closestCorners}
        onDragStart={handleDragStart}
        onDragEnd={handleDragEnd}
      >
        <div className="kanban-board">
          {COLUMNS.map((col) => (
            <KanbanColumn
              key={col.id}
              col={col}
              cards={cards.filter((c) => c.column === col.id)}
            />
          ))}
        </div>
        <DragOverlay>
          {activeCard && (
            <div className="kanban-card kanban-card-overlay">
              <span className="kanban-card-title">{activeCard.title}</span>
            </div>
          )}
        </DragOverlay>
      </DndContext>
    </div>
  );
}
```
